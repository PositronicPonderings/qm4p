# ---------------------------------------------------------------------------
#  Examples: each builds as its own program, build/examples/qg4p_<name>.uf2.
#  Program names start with the library they show off: qg4p_ for graphics.
#  They share board.c/board.h: set your wiring there once.
#  ASSETS: the example also uses an asset pack, so it links qa4p.
# ---------------------------------------------------------------------------
function(qg4p_example name)
    cmake_parse_arguments(E "ART;ASSETS" "" "" ${ARGN})
    add_executable(qg4p_${name} ${name}.c board.c)
    target_link_libraries(qg4p_${name} pico_stdlib qg4p)
    if (E_ART)
        target_sources(qg4p_${name} PRIVATE example_art.c)
    endif()
    if (E_ASSETS)
        target_link_libraries(qg4p_${name} qa4p)   # the asset pack library
    endif()
    pico_enable_stdio_usb(qg4p_${name} 1)
    pico_enable_stdio_uart(qg4p_${name} 0)
    # stdio_init_all() waits until a terminal connects, at most 2 seconds:
    # messages aren't lost, and without a terminal the program still starts.
    target_compile_definitions(qg4p_${name} PRIVATE PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=2000)
    pico_add_extra_outputs(qg4p_${name})
endfunction()
